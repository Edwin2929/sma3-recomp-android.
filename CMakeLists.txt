cmake_minimum_required(VERSION 3.22)
project(sma3_recomp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GBARECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../gbarecomp" CACHE PATH "GBARecomp checkout")
set(SMA3_PREBUILT_GAME "" CACHE FILEPATH "Optional same-platform generated game archive")
set(GBARECOMP_GENERATED_BIOS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated-bios" CACHE PATH "Recompiled user BIOS")
if(TRUE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/apply_interpolation_patch.py" "${GBARECOMP_ROOT}"
        RESULT_VARIABLE SMA3_PATCH_RESULT)
    if(NOT SMA3_PATCH_RESULT EQUAL 0)
        message(FATAL_ERROR "Cannot apply SMA3 interpolation patch to pinned engine")
    endif()
endif()
set(GBARECOMP_ENABLE_MODS ON CACHE BOOL "SMA3 validated color-profile packages" FORCE)
add_subdirectory("${GBARECOMP_ROOT}" engine EXCLUDE_FROM_ALL)
target_compile_definitions(gbarecomp_runtime PRIVATE SMA3_MOD_PROFILES=1)
target_compile_definitions(gbarecomp_runtime PRIVATE SMA3_LOCALIZATION=1)
target_include_directories(gbarecomp_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(SMA3_PREBUILT_GAME)
    if(ANDROID)
        if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
            message(FATAL_ERROR "The prebuilt Android archive requires arm64-v8a")
        endif()
        execute_process(COMMAND "${CMAKE_READELF}" -h "${SMA3_PREBUILT_GAME}"
            RESULT_VARIABLE archive_result OUTPUT_VARIABLE archive_headers)
        string(REGEX MATCHALL "Machine:[ ]+AArch64" archive_machines "${archive_headers}")
        list(LENGTH archive_machines archive_members)
        if(NOT archive_result EQUAL 0 OR NOT archive_members EQUAL 66)
            message(FATAL_ERROR "Expected 66 AArch64 objects in the prebuilt game archive")
        endif()
    elseif(CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "Unsupported prebuilt cross-compilation target")
    endif()
    add_library(sma3_game STATIC IMPORTED)
    set_target_properties(sma3_game PROPERTIES IMPORTED_LOCATION "${SMA3_PREBUILT_GAME}")
else()
    add_subdirectory(generated-project)
    add_library(sma3_game ALIAS gbarecomp_game)
    set_property(TARGET gbarecomp_game PROPERTY POSITION_INDEPENDENT_CODE ON)
endif()
if(ANDROID)
    add_library(main SHARED src/main.cpp)
    set(SMA3_TARGET main)
else()
    add_executable(sma3_runner src/main.cpp)
    set(SMA3_TARGET sma3_runner)
endif()
target_link_gbarecomp_runtime_stack(${SMA3_TARGET} sma3_game)
target_sources(${SMA3_TARGET} PRIVATE src/localization.cpp src/widescreen_bounds.cpp src/yoshi_color.cpp)
option(SMA3_PANORAMA "Build the optional experimental panoramic presentation" ON)
option(SMA3_WIDESCREEN_DIAGNOSTICS "Build desktop-only panoramic capture diagnostics" OFF)
if(SMA3_PANORAMA OR SMA3_WIDESCREEN_DIAGNOSTICS)
    target_sources(${SMA3_TARGET} PRIVATE src/widescreen.cpp)
endif()
if(SMA3_PANORAMA)
    target_compile_definitions(${SMA3_TARGET} PRIVATE SMA3_PANORAMA=1)
endif()
if(SMA3_WIDESCREEN_DIAGNOSTICS)
    if(ANDROID)
        message(FATAL_ERROR "Desktop capture diagnostics must not be packaged in Android")
    endif()
    target_compile_definitions(${SMA3_TARGET} PRIVATE SMA3_WIDESCREEN_DIAGNOSTICS=1)
endif()

file(GLOB SMA3_RAM_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated-ram/sma3ram_recompiled_*.cpp")
target_sources(${SMA3_TARGET} PRIVATE src/ram_dispatch.cpp ${SMA3_RAM_SOURCES} generated-ram/sma3ram_dispatch_table.cpp)
target_include_directories(${SMA3_TARGET} PRIVATE generated-ram)

file(GLOB SMA3_OVERLAY_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/generated-overlays/*/*recompiled_*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/generated-overlays/*/*dispatch_table.cpp")
target_sources(${SMA3_TARGET} PRIVATE src/overlay_dispatch.cpp ${SMA3_OVERLAY_SOURCES})

if(ANDROID)
    target_compile_definitions(gbarecomp_runtime PRIVATE SMA3_FRAME_INTERPOLATION=1)
    target_include_directories(gbarecomp_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_sources(main PRIVATE src/frame_interpolation_jni.cpp src/mod_profiles_jni.cpp)
endif()

file(GLOB SMA3_MORPH_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated-morph/morph_recompiled_*.cpp")
target_sources(${SMA3_TARGET} PRIVATE src/morph_dispatch.cpp ${SMA3_MORPH_SOURCES} generated-morph/morph_dispatch_table.cpp)

file(GLOB SMA3_EFFECT_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/generated-effects/*/*recompiled_*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/generated-effects/*/*dispatch_table.cpp")
if(NOT SMA3_EFFECT_SOURCES)
    message(FATAL_ERROR "Missing Fuzzy effect sources. Run tools/generate_effects.py with your verified ROM.")
endif()
target_sources(${SMA3_TARGET} PRIVATE src/effects_dispatch.cpp ${SMA3_EFFECT_SOURCES})

if(NOT ANDROID)
    add_executable(sma3_morph_test validation/morph_test.cpp src/localization.cpp src/morph_dispatch.cpp ${SMA3_MORPH_SOURCES} generated-morph/morph_dispatch_table.cpp)
    target_link_gbarecomp_runtime_stack(sma3_morph_test sma3_game)
    add_executable(sma3_effects_test validation/effects_test.cpp src/localization.cpp src/widescreen_bounds.cpp src/effects_dispatch.cpp ${SMA3_EFFECT_SOURCES})
    target_link_gbarecomp_runtime_stack(sma3_effects_test sma3_game)
endif()
