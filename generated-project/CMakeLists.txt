cmake_minimum_required(VERSION 3.22)
project(sma3_generated LANGUAGES CXX)
if(NOT DEFINED GBARECOMP_ROOT)
    set(GBARECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../gbarecomp")
endif()
file(GLOB SHARDS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated/recompiled_*.cpp")
if(NOT SHARDS)
    message(FATAL_ERROR "Run tools/generate_sources.py with your own ROM and BIOS first")
endif()
add_library(gbarecomp_game STATIC ${SHARDS} generated/dispatch_table.cpp generated/symbol_map.cpp)
target_include_directories(gbarecomp_game PRIVATE generated "${GBARECOMP_ROOT}/src/armv4t")
set_target_properties(gbarecomp_game PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE ON)
